# Scan feed Instagram & Facebook — rencana implementasi

> **Untuk agen yang mengerjakan:** kerjakan task berurutan. Setiap langkah memakai checkbox (`- [ ]`).
> Tulis tes dulu, jalankan sampai gagal, baru implementasi. Semua dependensi dipasang lokal di project
> (C: hampir penuh; tidak ada instalasi global).

**Tujuan:** Setiap postingan yang terlihat di feed IG/FB mendapat chip label dari scan teks murah
(`POST /v1/scan`, tanpa LLM). Mengklik chip menjalankan analisis penuh lewat `Panel` yang sudah ada.

**Arsitektur:** `FeedScanner` baru berjalan di samping `Controller` lama dan dipilih per halaman oleh content
script. Backend menambah `Pipeline.scan()` yang memakai `extract_claims` + `finder.quick` + `aggregate`
yang sudah ada.

**Stack:** FastAPI + pytest (`backend/`), WXT + React + Zustand + Vitest + Playwright (`extension/`).

**Spec:** `docs/superpowers/specs/2026-10-29-feed-scan-design.md`

```python
    async def scan(self, platform: str, items: list[ScanItem]) -> list[ScanResult]:
        """Label cepat postingan feed: cache hasil penuh atau heuristik + korpus. Tanpa LLM, sesi, dan sinyal penonton."""
        async def one(item: ScanItem) -> ScanResult:
            try:
                cached = await self.cache.get(result_key(platform, item.id))
                if cached is not None:
                    full = AnalysisResult(**cached)
                    return ScanResult(id=item.id, label=full.label, score=full.score, tni_related=full.tni_related,
                                      claims=len(full.claims), cached=True)
                claims = extract_claims(item.text, "caption", item.hashtags)[:MAX_SESSION_CLAIMS]
                factual = [c for c in claims if c.kind == "factual"]
                async with self._prefetch_slots:
                    found = await asyncio.gather(*(self.finder.quick(c) for c in factual))
                verdict = aggregate(claims, [e for ev in found for e in ev])
                return ScanResult(id=item.id, label=verdict.label, score=verdict.score,
                                  tni_related=any(c.tni_related for c in claims), claims=len(factual), cached=False)
            except Exception:
                log.exception("scan gagal untuk %s:%s", platform, item.id)
                return ScanResult(id=item.id, label="error", score=None, tni_related=False, claims=0, cached=False)

        return list(await asyncio.gather(*(one(i) for i in items)))
```

import type { BackgroundMessage } from './messages';
import { useOverlay } from './store';

/** Terapkan event SSE dari background ke overlay bila milik sesi yang sedang aktif. */
export function applySessionEvent(raw: unknown, activeKey: string | null | undefined): BackgroundMessage | null {
  const msg = raw as BackgroundMessage;
  if (msg?.type !== 'analyze:event' || !activeKey || msg.key !== activeKey) return null;
  const overlay = useOverlay.getState();
  if (overlay.key !== msg.key) return null;
  switch (msg.event) {
    case 'stage': overlay.stage(msg.data); break;
    case 'result': overlay.setResult(msg.data); break;
    case 'error': overlay.fail(msg.data.message); break;
    case 'done': overlay.done(); break;
  }
  return msg;
}
```

import type { ContentMessage } from '../messages';
import { AudioCapturer } from './audio';
import { FrameCapturer } from './frames';

export interface CaptureHandle {
  stop(): void;
}

/** Frame + audio satu video untuk sesi `key`; mengirim analyze:complete setelah keduanya selesai. */
export function captureVideo(opts: {
  key: string;
  video: HTMLVideoElement;
  accepts: { frames: boolean; audio: boolean };
  isActive: () => boolean;
  send: <T = unknown>(msg: ContentMessage) => Promise<T>;
}): CaptureHandle {
  const { key, video, accepts, isActive, send } = opts;
  let pending = Number(accepts.frames) + Number(accepts.audio);
  const finished = () => {
    if (--pending === 0 && isActive()) send({ type: 'analyze:complete', key });
  };
  const frames = new FrameCapturer(
    video,
    () => send<string | null>({ type: 'capture:tab' }),
    (shots) => {
      if (shots.length && isActive()) send({ type: 'analyze:frames', key, frames: shots });
      finished();
    },
  );
  const audio = new AudioCapturer(
    video,
    (chunk, index) => {
      if (isActive()) send({ type: 'analyze:audio', key, chunk, index });
    },
    () => finished(),
  );
  if (accepts.frames) frames.start();
  if (accepts.audio) audio.start();
  return {
    stop() {
      frames.stop();
      audio.stop();
    },
  };
}
```

const SPONSORED = /^(sponsored|bersponsor)$/i;

export function isSponsored(root: Element): boolean {
  for (const el of root.querySelectorAll('a, span')) {
    if (SPONSORED.test(el.textContent?.trim() ?? '')) return true;

/** ID dari tautan pertama di postingan yang cocok dengan pola (grup tangkap pertama yang terisi). */
export function firstLink(root: Element, pattern: RegExp): { id: string; url: string } | null {
  for (const a of root.querySelectorAll<HTMLAnchorElement>('a[href]')) {
    const href = a.getAttribute('href') ?? '';
    const id = href.match(pattern)?.slice(1).find(Boolean);
    if (id) return { id, url: new URL(href, location.origin).href };
  }
  return null;
}

function contentImages(root: Element): HTMLImageElement[] {
  return [...root.querySelectorAll('img')].filter((img) => !img.closest('header') && !/profil/i.test(img.alt));

  const video = root.querySelector('video');

  const img = contentImages(root)[0];
  return img ? { kind: 'image', media: img } : { kind: 'text', media: null };

import { create } from 'zustand';
import type { AnalysisResult, Label, Platform, ScanResult } from '../types';

export const MAX_ITEMS = 500; // feed tanpa ujung: hasil lama dibuang (LRU)

export type FeedStatus = 'queued' | 'done' | 'error';

export interface FeedEntry {
  status: FeedStatus;
  label: Label | null;
  score: number | null;
  tniRelated: boolean;
  at: number;
  retried: boolean;
}

/** Jembatan dari FeedScanner ke komponen chip. */
export interface FeedSource {
  platform: Platform;
  visible(): { id: string; anchor: Element }[];
  open(id: string): void;
}

interface FeedState {
  entries: Map<string, FeedEntry>;
  source: FeedSource | null;
  queued(ids: string[], now: number): void;
  resolved(results: ScanResult[], now: number): void;
  failed(ids: string[], now: number): void;
  final(id: string, result: AnalysisResult, now: number): void;
  setSource(source: FeedSource | null): void;
  clear(): void;
}

const EMPTY: Omit<FeedEntry, 'status' | 'at'> = { label: null, score: null, tniRelated: false, retried: false };

function upsert(entries: Map<string, FeedEntry>, updates: [string, FeedEntry][]): Map<string, FeedEntry> {
  const next = new Map(entries);
  for (const [id, entry] of updates) {
    next.delete(id); // sisipkan ulang agar menjadi yang terbaru
    next.set(id, entry);
  }
  for (const id of next.keys()) {
    if (next.size <= MAX_ITEMS) break;
    next.delete(id);
  }
  return next;
}

export const useFeed = create<FeedState>()((set) => ({
  entries: new Map(),
  source: null,
  queued: (ids, now) => set((s) => ({
    entries: upsert(s.entries, ids.map((id) => {
      const prev = s.entries.get(id);
      return [id, { ...(prev ?? EMPTY), status: 'queued', at: now, retried: prev?.status === 'error' }];
    })),
  })),
  resolved: (results, now) => set((s) => ({
    entries: upsert(s.entries, results.map((r) => [r.id, {
      status: r.label === 'error' ? 'error' : 'done', label: r.label === 'error' ? null : r.label,
      score: r.score, tniRelated: r.tni_related, at: now, retried: s.entries.get(r.id)?.retried ?? false,
    }])),
  })),
  failed: (ids, now) => set((s) => ({
    entries: upsert(s.entries, ids.map((id) => {
      const prev = s.entries.get(id);
      return [id, { ...(prev ?? EMPTY), status: 'error', at: now, retried: prev?.retried ?? false }];
    })),
  })),
  final: (id, result, now) => set((s) => ({
    entries: upsert(s.entries, [[id, {
      status: 'done', label: result.label, score: result.score, tniRelated: result.tni_related, at: now, retried: false,
    }]]),
  })),
  setSource: (source) => set({ source }),
  clear: () => set({ entries: new Map() }),
}));

```ts
import type { ScanItem, ScanResult } from '../types';
import { useFeed } from './store';

export const BATCH_SIZE = 10;
export const RETRY_AFTER_MS = 30_000;
export const PAUSE_AFTER_FAILURES = 3;
export const PAUSE_MS = 60_000;

type Send = (items: ScanItem[]) => Promise<ScanResult[] | null>;

/** Antrean scan feed: dedupe per ID, batch ≤10, satu percobaan ulang, jeda saat backend bermasalah. */
export class ScanQueue {
  private pending = new Map<string, ScanItem>();
  private failures = 0;
  private pausedUntil = 0;
  private inflight = false;

  constructor(private send: Send, private now: () => number = Date.now) {}

  enqueue(item: ScanItem) {
    const entry = useFeed.getState().entries.get(item.id);
    const retry = entry?.status === 'error' && !entry.retried && this.now() - entry.at >= RETRY_AFTER_MS;
    if (entry && !retry) return;
    this.pending.set(item.id, item);
  }

  async flush() {
    if (this.inflight || !this.pending.size || this.now() < this.pausedUntil) return;
    const batch = [...this.pending.values()].slice(0, BATCH_SIZE);
    for (const i of batch) this.pending.delete(i.id);
    const ids = batch.map((i) => i.id);
    useFeed.getState().queued(ids, this.now());
    this.inflight = true;
    try {
      const results = await this.send(batch);
      if (!results) throw new Error('scan gagal');
      this.failures = 0;
      useFeed.getState().resolved(results, this.now());
      const missing = ids.filter((id) => !results.some((r) => r.id === id));
      if (missing.length) useFeed.getState().failed(missing, this.now());
    } catch {
      useFeed.getState().failed(ids, this.now());
      if (++this.failures >= PAUSE_AFTER_FAILURES) {
        this.failures = 0;
        this.pausedUntil = this.now() + PAUSE_MS;
      }
    } finally {
      this.inflight = false;
    }
  }

  clear() {
    this.pending.clear();
  }
}

import type { FeedAdapter, FeedItem } from './types';

const TICK_MS = 400;
const FLUSH_MS = 500;
const DWELL_MS = 400;   // postingan yang hanya dilewati saat scroll cepat tidak discan
const VISIBLE = 0.4;

const toScanItem = ({ id, kind, text, hashtags, url }: FeedItem): ScanItem => ({ id, kind, text, hashtags, url });
const nextFrame = () => new Promise((r) => requestAnimationFrame(() => requestAnimationFrame(r)));

interface Active {
  key: string;
  id: string;
  capture?: CaptureHandle;
}

/** Scan teks semua postingan feed yang terlihat; analisis penuh hanya untuk postingan yang chip-nya diklik. */
export class FeedScanner {
  private items = new Map<string, FeedItem>();          // elemen terkini per ID
  private visibleSince = new Map<Element, number>();    // ≥40% terlihat sejak
  private onScreen = new Set<Element>();                // terlihat sedikit pun (untuk posisi chip)
  private observer: IntersectionObserver;
  private queue: ScanQueue;
  private active: Active | null = null;
  private enabled = false;
  private mode: Mode = 'multimodal';
  private timers: number[] = [];
  private unwatch: (() => void)[] = [];

  constructor(private adapter: FeedAdapter) {
    this.observer = new IntersectionObserver((entries) => {
      const now = performance.now();
      for (const e of entries) {
        if (e.isIntersecting) this.onScreen.add(e.target);
        else this.onScreen.delete(e.target);
        if (e.intersectionRatio >= VISIBLE) this.visibleSince.set(e.target, this.visibleSince.get(e.target) ?? now);
        else this.visibleSince.delete(e.target);
      }
    }, { threshold: [0, VISIBLE, 1] });
    this.queue = new ScanQueue(async (items) => {
      const res = await send<ScanResponse>({ type: 'feed:scan', platform: adapter.platform, items }).catch(() => null);
      return res?.ok ? res.results : null;
    });
  }

  async start() {
    await this.refreshSettings();
    this.unwatch.push(
      enabledItem.watch(() => this.refreshSettings()),
      platformsItem.watch(() => this.refreshSettings()),
      feedEnabledItem.watch(() => this.refreshSettings()),
      modeItem.watch((m) => (this.mode = m)),
    );
    browser.runtime.onMessage.addListener(this.onMessage);
    this.timers.push(
      window.setInterval(() => this.tick(), TICK_MS),
      window.setInterval(() => void this.queue.flush(), FLUSH_MS),
    );
    useFeed.getState().setSource({
      platform: this.adapter.platform,
      visible: () => [...this.items.values()].filter((i) => this.onScreen.has(i.root)).map((i) => ({ id: i.id, anchor: i.anchor })),
      open: (id) => void this.open(id),
    });
  }

  destroy() {
    this.close();
    this.timers.forEach((t) => window.clearInterval(t));
    this.unwatch.forEach((fn) => fn());
    browser.runtime.onMessage.removeListener(this.onMessage);
    this.observer.disconnect();
    this.queue.clear();
    useFeed.getState().setSource(null);
    useFeed.getState().clear();
  }

  private async refreshSettings() {
    const [enabled, platforms, feed, mode] = await Promise.all([
      enabledItem.getValue(), platformsItem.getValue(), feedEnabledItem.getValue(), modeItem.getValue(),
    ]);
    this.enabled = enabled && feed && platforms[this.adapter.platform] !== false;
    this.mode = mode;
    send({ type: 'status', active: this.enabled }).catch(() => {});
    if (!this.enabled) {
      this.close();
      this.queue.clear();
      useFeed.getState().clear();
    }
  }

  private tick() {
    if (!this.enabled) return;
    const now = performance.now();
    for (const item of this.adapter.findItems()) {
      const prev = this.items.get(item.id);
      if (prev?.root !== item.root) {
        // Feed merender ulang postingan yang sama: pindahkan observasi ke elemen baru.
        if (prev) this.forget(prev.root);
        this.observer.observe(item.root);
      }
      this.items.delete(item.id);
      this.items.set(item.id, item);
      const since = this.visibleSince.get(item.root);
      if (since !== undefined && now - since >= DWELL_MS) this.queue.enqueue(toScanItem(item));
    }
    for (const [id, item] of this.items) {
      if (this.items.size <= MAX_ITEMS) break;
      this.forget(item.root);
      this.items.delete(id);
    }
  }

  private forget(root: Element) {
    this.observer.unobserve(root);
    this.visibleSince.delete(root);
    this.onScreen.delete(root);
  }

  private close() {
    if (!this.active) return;
    this.active.capture?.stop();
    send({ type: 'analyze:cancel', key: this.active.key }).catch(() => {});
    this.active = null;
    useOverlay.getState().clear();
  }

  async open(id: string) {
    const item = this.items.get(id);
    if (!item) return;
    this.close();
    const key = `${this.adapter.platform}:${id}`;
    const active: Active = { key, id };
    this.active = active;
    const isActive = () => this.active === active;
    const mode: Mode = item.kind === 'text' ? 'caption' : this.mode;

    // Foto diambil sebelum panel terbuka agar screenshot tab tidak tertutup panel.
    let frame: string | null = null;
    if (item.kind === 'image' && mode === 'multimodal') {
      await nextFrame();
      const img = visibleImage(item.root);
      frame = img && (await captureElement(img, () => send<string | null>({ type: 'capture:tab' })));
      if (!isActive()) return;
    }

    const overlay = useOverlay.getState();
    overlay.begin({
      key, platform: this.adapter.platform, videoId: id,
      geometry: () => {
        const cur = this.items.get(id);
        if (!cur?.root.isConnected) return null;
        return { anchor: cur.anchor.getBoundingClientRect(), video: cur.root.getBoundingClientRect() };
      },
      retry: () => void this.open(id),
    });
    overlay.togglePanel(true);

    let res: StartResponse;
    try {
      res = await send<StartResponse>({
        type: 'analyze:start', key,
        payload: { platform: this.adapter.platform, video_id: id, url: item.url, caption: item.text, hashtags: item.hashtags, mode },
      });
    } catch (err) {
      res = { ok: false, error: String(err) };
    }
    if (!isActive()) return;
    if (!res.ok) return useOverlay.getState().fail(res.error);
    if (res.cached) {
      useOverlay.getState().setResult(res.result);
      useFeed.getState().final(id, res.result, Date.now());
      return;

    if (mode === 'multimodal' && item.kind === 'video' && item.media instanceof HTMLVideoElement

      return;

    if (item.kind === 'image' && mode === 'multimodal' && res.accepts.frames) {
      if (frame) send({ type: 'analyze:frames', key, frames: [frame] }).catch(() => {});
      else useOverlay.getState().note('Gambar tidak dapat dibaca.');
    }
    send({ type: 'analyze:complete', key }).catch(() => {});
  }

      useFeed.getState().final(active.id, msg.data, Date.now());
    }
  };
}

      for (const { id, anchor } of source.visible()) {
        const r = anchor.getBoundingClientRect();
        if (r.width === 0 || r.bottom < 0 || r.top > innerHeight) continue;
        next.push([id, {
          left: Math.min(Math.max(r.left, EDGE), innerWidth - CHIP_WIDTH - EDGE),
          top: Math.max(EDGE, r.top - CHIP_HEIGHT - GAP),
        }]);
      }
      const sig = JSON.stringify(next);
      if (sig !== last) {
        last = sig;
        setPositions(next);
      }
      frame = requestAnimationFrame(loop);
    };
    frame = requestAnimationFrame(loop);
    return () => cancelAnimationFrame(frame);
  }, [source]);

  if (!source) return null;
  return (
    <div className="nr-layer">
      {positions.map(([id, p]) => {
        const entry = entries.get(id);
        if (!entry || activeKey === `${source.platform}:${id}`) return null;
        return (
          <div key={id} className="nr-anchor" style={{ left: p.left, top: p.top }}>
            <FeedChip id={id} entry={entry} onOpen={() => source.open(id)} />
          </div>
        );
      })}
    </div>
  );
}

function FeedChip({ id, entry, onOpen }: { id: string; entry: FeedEntry; onOpen: () => void }) {
  if (entry.status === 'queued') {
    return (
      <button className="nr-chip nr-chip--queued nr-chip--dim" data-item-id={id} aria-label="NARASI: sedang memeriksa" onClick={onOpen}>
        <span className="nr-dot" aria-hidden />
      </button>
    );
  }
  if (entry.status === 'error' || !entry.label) {
    return (
      <button className="nr-chip nr-chip--dim" data-item-id={id} aria-label="NARASI: belum bisa diperiksa. Klik untuk analisis lengkap." onClick={onOpen}>
        –
      </button>
    );
  }
  const label = LABELS[entry.label];
  const dim = DIM.includes(entry.label);
  const mencatut = entry.tniRelated && (entry.label === 'hoax' || entry.label === 'caution');
  const score = !dim && label.showScore && entry.score !== null ? `${entry.score}%` : null;
  return (
    <button
      className={`nr-chip${dim ? ' nr-chip--dim' : ''}`}
      data-item-id={id}
      style={{ '--nr-color': label.color } as React.CSSProperties}
      aria-label={`NARASI: ${label.text}${score ? ` ${score}` : ''}. Klik untuk analisis lengkap.`}
      onClick={onOpen}
    >
      <span className="nr-dot" aria-hidden />
      <span>{label.text}</span>
      {score && <strong>{score}</strong>}
      {mencatut && <span className="nr-tag">Mencatut TNI</span>}
    </button>
  );
}
```
